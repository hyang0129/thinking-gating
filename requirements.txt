# Dependencies for the thinking-gating repo. Self-contained: everything the
# scripts import is listed here, and nothing is expected from a sibling repo.
#
# Install into this repo's own virtualenv (see scripts/setup_env.sh):
#     python -m venv .venv && . .venv/bin/activate && pip install -r requirements.txt
#
# The System One work (Qwen3.5 / Kev / vLLM / peft) needs a newer transformers
# than the <5 pin below and gets its own environment (requirements-s1.txt, not
# created yet). Do not upgrade this one in place: the cluster redo captures run
# in it.

# --- Core ML -----------------------------------------------------------------
# transformers >= 4.51 is a hard floor: Qwen3 support and the `enable_thinking`
# chat-template flag that the whole paired-capture design rests on both landed
# there. Older pins (4.40) cannot load Qwen/Qwen3-8B at all.
torch>=2.4
# Upper bound is deliberate: transformers 5.x renames `torch_dtype` (used in
# scripts/capture_inference_thinking.py) and reworks generation internals, so
# stay on the 4.x line until that migration is done on purpose.
transformers>=4.51,<5
accelerate>=0.30          # required by device_map="auto" in the capture script
numpy>=1.26,<2.0
scikit-learn>=1.4         # probes, TF-IDF baselines, AUROC (also brings scipy)

# --- Data --------------------------------------------------------------------
datasets>=2.19,<5         # tasks/*.py loaders (parquet-only); pulls in pandas itself

# --- Dispatch tooling (scripts/gpu_dispatch.py -> utils/jupyter_exec.py) ------
requests>=2.31
websocket-client>=1.7

# --- Development -------------------------------------------------------------
# tests/test_graders.py also checks MATH-500 grading against `math_verify` when
# it is installed; it is optional and skipped otherwise.
pytest>=7.4
black>=23.12
ruff>=0.1.9
